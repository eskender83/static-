import os import json from pathlib import Path from decimal import Decimal, InvalidOperation import httpx from fastapi import FastAPI, HTTPException from fastapi.middleware.cors import CORSMiddleware from pydantic import BaseModel, Field # ========================================== # НАСТРОЙКИ # ========================================== BASE_DIR = Path(__file__).resolve().parent DATA_FILE = BASE_DIR / "data.json" # Переменные берутся из Render Environment BOT_TOKEN = os.getenv("BOT_TOKEN") CHAT_ID = os.getenv("CHAT_ID") print("BOT_TOKEN установлен:", bool(BOT_TOKEN)) print("CHAT_ID установлен:", bool(CHAT_ID)) # ========================================== # СОЗДАНИЕ ПРИЛОЖЕНИЯ # ========================================== app = FastAPI(title="Auto Parts Shop API") app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=False, allow_methods=["*"], allow_headers=["*"], ) # ========================================== # ЗАГРУЗКА ТОВАРОВ # ========================================== def load_products(): if not DATA_FILE.exists(): raise HTTPException( status_code=500, detail="Файл data.json не найден" ) try: with open(DATA_FILE, "r", encoding="utf-8") as file: products = json.load(file) except (json.JSONDecodeError, OSError) as e: print("Ошибка чтения data.json:", str(e)) raise HTTPException( status_code=500, detail="Ошибка чтения каталога товаров" ) if not isinstance(products, list): raise HTTPException( status_code=500, detail="Неверный формат data.json" ) return products # ========================================== # ГЛАВНАЯ СТРАНИЦА # ========================================== @app.get("/") async def home(): return { "status": "OK", "message": "Auto Parts Shop API работает" } # ========================================== # ПОЛУЧЕНИЕ КАТАЛОГА # ========================================== @app.get("/api/products") async def get_products(): return load_products() # ========================================== # МОДЕЛЬ ТОВАРА В ЗАКАЗЕ # ========================================== class CartItem(BaseModel): id: str quantity: int = Field(gt=0, le=1000) # ========================================== # МОДЕЛЬ ЗАКАЗА # ========================================== class OrderRequest(BaseModel): user_id: str | int = "Не указан" user_name: str = "Покупатель" phone: str vin: str = "" delivery: str = "" items: list[CartItem] total_price: str | int | float | None = None # ========================================== # ПРЕОБРАЗОВАНИЕ ЦЕНЫ # ========================================== def parse_price(value): try: clean_value = ( str(value) .replace(" ", "") .replace("\u00a0", "") .replace(",", ".") ) return Decimal(clean_value) except (InvalidOperation, ValueError, TypeError): raise HTTPException( status_code=500, detail=f"Неверная цена товара: {value}" ) def format_price(value): return f"{value:,.2f}".replace(",", " ").replace(".00", "") # ========================================== # ФОРМИРОВАНИЕ СООБЩЕНИЯ О ЗАКАЗЕ # ========================================== def create_order_message(order, products): products_by_id = { str(product.get("id")): product for product in products } message = [ "🛒 НОВЫЙ ЗАКАЗ", "", f"👤 Покупатель: {order.user_name}", f"🆔 Telegram ID: {order.user_id}", f"📞 Телефон: {order.phone}", f"🚗 VIN: {order.vin or 'Не указан'}", f"📦 Доставка: {order.delivery or 'Не указана'}", "", "ТОВАРЫ:" ] total = Decimal("0") for item in order.items: product = products_by_id.get(str(item.id)) if not product: raise HTTPException( status_code=400, detail=f"Товар с ID {item.id} не найден" ) title = product.get("title", "Без названия") oem = product.get("oem", "") price = parse_price(product.get("price", 0)) subtotal = price * item.quantity total += subtotal message.append("") message.append(f"• {title}") if oem: message.append(f" Артикул: {oem}") message.append(f" Цена: {format_price(price)} грн") message.append(f" Количество: {item.quantity}") message.append(f" Сумма: {format_price(subtotal)} грн") message.extend([ "", "━━━━━━━━━━━━━━━━", f"💰 ИТОГО: {format_price(total)} грн" ]) return "\n".join(message) # ========================================== # ОТПРАВКА СООБЩЕНИЯ В TELEGRAM # ========================================== async def send_telegram_message(message: str): if not BOT_TOKEN or not CHAT_ID: print("Ошибка: BOT_TOKEN или CHAT_ID не настроены") raise HTTPException( status_code=500, detail="Не настроены BOT_TOKEN или CHAT_ID в Render" ) url = ( f"https://api.telegram.org/" f"bot{BOT_TOKEN}/sendMessage" ) payload = { "chat_id": CHAT_ID, "text": message } try: async with httpx.AsyncClient(timeout=30.0) as client: response = await client.post( url, json=payload ) print( "Ответ Telegram HTTP:", response.status_code ) if response.status_code != 200: print( "Ошибка Telegram:", response.text ) response.raise_for_status() result = response.json() if not result.get("ok"): print("Ошибка Telegram API:", result) raise HTTPException( status_code=502, detail=f"Telegram API: {result}" ) print("Заказ успешно отправлен в Telegram") return result except HTTPException: raise except httpx.HTTPStatusError as e: print( "ОШИБКА TELEGRAM:", e.response.status_code ) print( "ОТВЕТ TELEGRAM:", e.response.text ) raise HTTPException( status_code=502, detail=f"Ошибка Telegram API: {e.response.text}" ) except httpx.RequestError as e: print( "ОШИБКА СОЕДИНЕНИЯ С TELEGRAM:", repr(e) ) raise HTTPException( status_code=502, detail="Не удалось подключиться к Telegram API" ) # ========================================== # ОФОРМЛЕНИЕ ЗАКАЗА # ========================================== @app.post("/api/order") async def create_order(order: OrderRequest): if not order.items: raise HTTPException( status_code=400, detail="Корзина пуста" ) if not order.phone.strip(): raise HTTPException( status_code=400, detail="Укажите номер телефона" ) products = load_products() message = create_order_message( order, products ) await send_telegram_message(message) return { "success": True, "message": "Заказ успешно отправлен" }
